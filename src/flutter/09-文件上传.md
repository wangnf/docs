
1. 安装 file_picker
```bash
flutter pub get file_picker
```

2. 获取文件
```dart
import 'package:file_picker/file_picker.dart';

Future<PlatformFile?> pickFile() {
    FilePickerResult? result = await FilePicker.pickFiles(
      allowedExtensions: ["gz"], // 限制 gz 后缀的文件可选
      allowMultiple: false, // 是否可多选
      type: FileType.custom, // 文件类型限制
    );
    if (result == null) return null;
    final file = result.files.first;
    if (!file.name.endsWith(".tar.gz")) return null;
    return file;
}

```

3. http文件上传
```dart
<!-- ********************** Component *********************** -->
import 'package:file_picker/file_picker.dart';
import 'package:dio/dio.dart';

int _progress = 0;
Future<void> uploadFile(PlatformFile platformFile) {
    // 判断文件路径是否正确
    if(platformFile.path == null) return;
    final file = File(platformFile.path!);

    final resp = await UpgradeApi.uploadUpgradePackage(
        file: file,
        onSendProgress: (sent, total) {
            if (total <= 0) return;
            setState(() {
                _progress = sent / total;
            });
        },
    );
}

<!-- ******************** API ************************* -->
import 'package:dio/dio.dart';

class UpgradeApi {

    static Future<UploadRes> uploadUpgradePackage(
        required File file,
        Function(int sent, int total)? onSendProgress,
    ) {
        final dio = Dio();
        final fileName = file.path.split('/').last;
        final fileLength = await file.length();

        final stream = file.openRead();
        final response = await dio.post(
            'http://$robotIp:9000/api/v1/upgrade/upload',
            // 直接传二进制流
            data: stream,
            options: Options(
                headers: {
                    'Authorization': 'Bearer $token',
                    'X-Filename': fileName,
                    'Content-Type': 'application/octet-stream',
                    'Content-Length': fileLength,
                },
                // 大文件建议关闭超时或者调大
                sendTimeout: const Duration(minutes: 3),
                receiveTimeout: const Duration(minutes: 3),
            ),
            onSendProgress: onSendProgress,
        );
    }
}
```
