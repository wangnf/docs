1. auto

int main() {
    auto a = 1;
    auto b = 1.0;
    auto c = 1.1f;
    auto d = true;
    auto e = "e";
    auto f = 'f';

    std::cout << typeid(a).name() << std::endl;
    std::cout << typeid(b).name() << std::endl;


    return 0;
}

3. 自增 自减
```c++
#include <iostream>

int main() {
    int a = 100;
    int b = 10;

    std::cout << a++ << std::endl;
    std::cout << ++a << std::endl;
    std::cout << b-- << std::endl;
    std::cout << --a << std::endl;
    
    return 0;
}
```

4. 预处理器指令
```c++
#include <iostream>

#define MAX_AGE 100
#define DEBUG

int main() {

#ifdef DEBUG
std::cout << "xxxx" << std::endl;
#endif

    int result = MAX_AGE / 10;
    std::cout << result << std::endl;

    return 0;
}
```
